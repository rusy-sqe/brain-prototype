04 · Usage
// Import from compiled bundle
// const { Icon } = window.SQESuiteDesignSystem_2c1e79;

// Default — solid
<Icon name="home" />

// Variant
<Icon name="user" variant="regular" />
<Icon name="wallet" variant="light" />
<Icon name="github" variant="brands" />

// Size + color
<Icon name="bolt" size="2x" color="var(--hz-horizon-primary)" />

// Fixed-width (align icons in nav lists)
<Icon name="home" fw />

// Loading state
<Icon name="spinner" spin />

// Accessible (decorative icons need no label)
<Icon name="download" aria-label="Download file" />
